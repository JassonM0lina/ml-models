# Function to get current month's weather data
def get_august_2025_weather():
    """Fetch weather data for August 2025 so far"""
    cities = {
        'Riyadh': {'lat': 24.7136, 'lon': 46.6753, 'weight': 0.35},
        'Jeddah': {'lat': 21.5433, 'lon': 39.1728, 'weight': 0.25},
        'Dammam': {'lat': 26.3927, 'lon': 49.9777, 'weight': 0.20},
        'Mecca': {'lat': 21.4225, 'lon': 39.8262, 'weight': 0.20}
    }
    
    start_date = '2025-08-01'
    end_date = '2025-08-27'  # Today
    
    weather_stats = {}
    
    for city, info in cities.items():
        try:
            response = requests.get(
                "https://archive-api.open-meteo.com/v1/archive",
                params={
                    'latitude': info['lat'],
                    'longitude': info['lon'],
                    'start_date': start_date,
                    'end_date': end_date,
                    'daily': ['temperature_2m_max', 'temperature_2m_min', 'temperature_2m_mean',
                             'relative_humidity_2m_mean'],
                    'timezone': 'Asia/Riyadh'
                },
                timeout=30
            )
            
            if response.status_code == 200:
                data = response.json()['daily']
                temps_max = [t for t in data['temperature_2m_max'] if t is not None]
                temps_mean = [t for t in data['temperature_2m_mean'] if t is not None]
                humidity = [h for h in data['relative_humidity_2m_mean'] if h is not None]
                
                weather_stats[city] = {
                    'weight': info['weight'],
                    'temp_max_mean': np.mean(temps_max),
                    'temp_mean_mean': np.mean(temps_mean),
                    'extreme_heat_days': sum(1 for t in temps_max if t > 45),
                    'very_hot_days': sum(1 for t in temps_max if t > 42),
                    'hot_days': sum(1 for t in temps_max if t > 40),
                    'humidity_mean': np.mean(humidity),
                    'cooling_degree_days': sum(max(0, t - 24) for t in temps_mean)
                }
                print(f"✓ {city}: Avg Max Temp = {weather_statsonyaity]['temp_max_mean']:.1f}°C")
        except Exception as e:
            print(f"Error fetching {city}: {e}")
    
    # Calculate weighted averages
    if weather_stats:
        total_weight = sum(w['weight'] for w in weather_stats.values())
        
        weighted_stats = {
            'temp_max_mean': sum(w['temp_max_mean'] * w['weight'] for w in weather_stats.values()) / total_weight,
            'temp_mean_mean': sum(w['temp_mean_mean'] * w['weight'] for w in weather_stats.values()) / total_weight,
            'extreme_heat_sum': sum(w['extreme_heat_days'] * w['weight'] for w in weather_stats.values()) / total_weight * 31/27,  # Scale to full month
            'very_hot_sum': sum(w['very_hot_days'] * w['weight'] for w in weather_stats.values()) / total_weight * 31/27,
            'hot_sum': sum(w['hot_days'] * w['weight'] for w in weather_stats.values()) / total_weight * 31/27,
            'humidity_mean_mean': sum(w['humidity_mean'] * w['weight'] for w in weather_stats.values()) / total_weight,
            'cooling_degree_days_24_sum': sum(w['cooling_degree_days'] * w['weight'] for w in weather_stats.values()) / total_weight * 31/27
        }
        
        return weighted_stats
    
    return None

def prepare_monthly_features(weather_daily, target_year, target_month):
    """Prepare features for a specific month"""
    
    # Filter for target month
    month_weather = weather_daily[
        (weather_daily['date'].dt.year == target_year) & 
        (weather_daily['date'].dt.month == target_month)
    ]
    
    if len(month_weather) == 0:
        return None
    
    # Aggregate to monthly features
    features = {}
    
    # Basic aggregations
    features['temp_max_mean_mean'] = month_weather['temp_max_mean'].mean()
    features['temp_max_mean_std'] = month_weather['temp_max_mean'].std()
    features['temp_max_max_max'] = month_weather['temp_max_max'].max()
    features['temp_max_max_mean'] = month_weather['temp_max_max'].mean()
    features['temp_mean_mean_mean'] = month_weather['temp_mean_mean'].mean()
    features['temp_mean_mean_max'] = month_weather['temp_mean_mean'].max()
    features['temp_mean_std_mean'] = month_weather['temp_mean_std'].mean()
    features['humidity_mean_mean'] = month_weather['humidity_mean'].mean()
    features['humidity_min_min'] = month_weather['humidity_min'].min()
    features['wind_speed_mean'] = month_weather['wind_speed'].mean()
    features['apparent_max_max_max'] = month_weather['apparent_max_max'].max()
    
    # Cooling degree days
    for base in [22, 24, 26, 28]:
        features[f'cdd_{base}_sum'] = np.maximum(0, month_weather['temp_mean_mean'] - base).sum()
    
    # Extreme heat counts
    features['extreme_48_sum'] = (month_weather['temp_max_max'] > 48).sum()
    features['extreme_46_sum'] = (month_weather['temp_max_max'] > 46).sum()
    features['extreme_44_sum'] = (month_weather['temp_max_max'] > 44).sum()
    features['extreme_42_sum'] = (month_weather['temp_max_max'] > 42).sum()
    
    # Heat index approximation
    T = month_weather['temp_mean_mean'].mean()
    RH = month_weather['humidity_mean'].mean()
    features['heat_index_mean'] = -8.78 + 1.61*T + 2.34*RH - 0.146*T*RH
    features['heat_index_max'] = features['heat_index_mean'] + 5  # Approximate
    
    # Heat streak (simplified - max consecutive hot days)
    hot_days = (month_weather['temp_max_mean'] > 42).astype(int)
    streaks = hot_days.groupby((hot_days != hot_days.shift()).cumsum()).sum()
    features['heat_streak_max'] = streaks.max() if len(streaks) > 0 else 0
    
    # Temporal features
    features['year'] = target_year
    features['month'] = target_month
    features['is_summer'] = 1 if target_month in [6, 7, 8, 9] else 0
    features['is_peak_summer'] = 1 if target_month in [7, 8] else 0
    features['months_since_start'] = (target_year - 2009) * 12 + target_month
    features['year_scaled'] = (target_year - 2009) / 10
    features['gas_availability_proxy'] = features['year_scaled'] * (-50)
    features['is_ramadan_summer'] = 0  # Would need actual dates
    
    # Lag features from historical data
    if 'df_historical' in globals():
        recent_burns = df_historical['direct_burn_kbd'].values[-12:]
        if len(recent_burns) >= 12:
            features['burn_lag1'] = recent_burns[-1]
            features['burn_lag2'] = recent_burns[-2] if len(recent_burns) >= 2 else recent_burns[-1]
            features['burn_lag3'] = recent_burns[-3] if len(recent_burns) >= 3 else recent_burns[-1]
            features['burn_lag12'] = recent_burns[0]
            features['burn_ma2'] = np.mean(recent_burns[-2:])
            features['burn_ma3'] = np.mean(recent_burns[-3:])
            features['burn_ma6'] = np.mean(recent_burns[-6:])
            features['burn_yoy'] = recent_burns[-1] / recent_burns[0] - 1 if recent_burns[0] > 0 else 0
        
        recent_temps = df_historical['temp_max_mean'].values[-12:] if 'temp_max_mean' in df_historical.columns else None
        if recent_temps is not None and len(recent_temps) >= 12:
            features['temp_lag1'] = recent_temps[-1]
            features['temp_lag2'] = recent_temps[-2] if len(recent_temps) >= 2 else recent_temps[-1]
            features['temp_lag3'] = recent_temps[-3] if len(recent_temps) >= 3 else recent_temps[-1]
            features['temp_lag12'] = recent_temps[0]
            features['temp_ma2'] = np.mean(recent_temps[-2:])
            features['temp_ma3'] = np.mean(recent_temps[-3:])
            features['temp_ma6'] = np.mean(recent_temps[-6:])
            features['temp_yoy'] = recent_temps[-1] - recent_temps[0]
    
    # Interaction features
    features['summer_intensity'] = features['is_summer'] * features['temp_max_max_max']
    features['peak_heat'] = features['is_peak_summer'] * features['extreme_44_sum']
    features['dry_heat'] = features['temp_max_max_max'] * (100 - features['humidity_min_min']) / 100
    features['temp_above_45'] = max(0, features['temp_max_max_max'] - 45) ** 2
    features['temp_above_43'] = max(0, features['temp_max_max_max'] - 43) ** 1.5
    features['cumulative_cdd'] = features['cdd_24_sum'] * 1.5  # Simplified
    features['temp_regional_spread'] = features['temp_mean_std_mean']
    
    return features
